# CH5xx 项目模板 - 根 CMakeLists
# ---------------------------------------------------------------------------
# 目录结构：
#   <项目根>/
#   ├── CMakeLists.txt        <- 本文件
#   ├── sdk_config.cmake      <- ★ 芯片选择（CH572 / CH592 / CH591）
#   ├── main/                 <- 应用代码（多芯片共用）
#   │   └── usb_dual/
#   ├── sdk/                  <- SDK 集合
#   │   ├── common/           <- 共享：工具链、ch32fun、USB 栈、构建逻辑
#   │   ├── CH572_DEV/        <- CH572 专属（BLE 库）
#   │   └── CH592_DEV/        <- CH592/CH591 专属（BLE 库）
#   ├── tool/                 <- WCH 烧录工具
#   └── tmp/                  <- 参考源码（gitignore）
#
# 构建：
#   cmake -B build -G Ninja -DCMAKE_TOOLCHAIN_FILE=sdk/common/cmake/riscv.cmake
#   cmake --build build
#
# 切换芯片：
#   修改 default_target.txt 里的 chip，或
#   cmake -B build -DCHIP_SDK=CH592 ...
#
# 产物在 build/main/<目标名>/ 下。
# ---------------------------------------------------------------------------

cmake_minimum_required(VERSION 3.20)
project(ch5xx_template C ASM)

# ---------------------------------------------------------------------------
# 芯片选择
# ---------------------------------------------------------------------------
# -DCHIP_SDK=... 为命令行覆盖（一次性，见 sdk_config.cmake 的说明）。
include(${CMAKE_CURRENT_SOURCE_DIR}/sdk_config.cmake)

# ---------------------------------------------------------------------------
# 默认烧录目标（可选）
# ---------------------------------------------------------------------------
# 决定 `./build flash` / `size` / `verify` 等裸目标作用于哪个固件。
# 优先级：-DCH572_TARGET=xxx  >  环境变量  >  default_target.txt  >  第一个固件
#
# 最便携的方式：修改项目根的 default_target.txt（第一行写固件名）。

# ---------------------------------------------------------------------------
# 引入 SDK（作为子项目）
# ---------------------------------------------------------------------------
add_subdirectory(sdk)

# ---------------------------------------------------------------------------
# 应用
# ---------------------------------------------------------------------------
add_subdirectory(main)

# 校验默认目标是否存在（必须在所有固件注册之后）
ch572_finalize()
